\renewcommand{\maketitle}{%
    \begin{titlepage}
        \centering
        \vspace*{0.2\textheight}
        {\fontsize{50}{48}\selectfont\bfseries\@title\par}
        \vspace{1em}
        \ifx\@subtitle\@empty\else
            {\LARGE\@subtitle\par}
        \fi
        \vfill
        % {\large\@author\par}
        % \vspace{2em}
        {\large\@date\par}
    \end{titlepage}%
}
