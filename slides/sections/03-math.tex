\begin{frame}[fragile]
    \frametitle{Math Mode}
    \begin{center}
        \Large
        \renewcommand{\arraystretch}{1.5}
        \begin{tabular}{@{}l@{\hspace{3em}}l@{\hspace{3em}}l@{}}
            \toprule
                                          & \textbf{Inline}              & \textbf{Display} \\
            \midrule
            \onslide<1->\textbf{\LaTeX}   & \onslide<1->\code|\(...\)|   & \onslide<1->\code|\[...\]| \\
            \onslide<2->\textbf{Plain \TeX} & \onslide<2->\code|$...$|   & \onslide<2->\code|$$...$$| \\
            \bottomrule
        \end{tabular}
    \end{center}
    \vfill
    \begin{uncoverenv}<3->
        \begin{center}
            \color{black!60}
            \code|$...$| does the same as \code|\(...\)|, and is fine to use
        \end{center}
    \end{uncoverenv}
    \vfill
    \begin{alertblock}<4->{\texttt{\$\$...\$\$}}
        Avoid it: not \LaTeX{} syntax, uneven spacing, no \texttt{amsmath} support.
        Use \texttt{\textbackslash[...\textbackslash]} instead.
    \end{alertblock}
    \note{%
        Inline: inside a sentence.
        Display: on its own, centred line.
        \begin{itemize}
            \item \texttt{\$...\$} is what everyone writes, the template included.
                \texttt{\textbackslash(...\textbackslash)} gives the clearer error message when one side is missing,
                because opening and closing are different characters.
            \item Why not \texttt{\$\$...\$\$}:
                it is plain \TeX{}, which \LaTeX{} never adapted.
                \texttt{\textbackslash[} is \LaTeX{}'s own, and \texttt{amsmath} builds on it.
            \item In practice: the space around the formula can come out wrong
                (e.g.\ an extra empty line above it after a paragraph break),
                the class option \texttt{fleqn} (left-aligned equations) ignores it,
                and \texttt{amsmath} features such as \texttt{\textbackslash tag} do not work inside it.
            \item A missing \texttt{\textbackslash]} is a clear error;
                a missing \texttt{\$\$} silently flips the rest of the text in or out of math mode.
            \item They \emph{will} have met \texttt{\$\$} in Markdown, Jupyter and on StackExchange.
                It works there because those are not \LaTeX{}.
                Say it once, clearly; the handout repeats it.
            \item Also exists: \texttt{equation*} from \texttt{amsmath}, the same as \texttt{\textbackslash[...\textbackslash]}.
        \end{itemize}
    }
\end{frame}

    \note{%
        The two characters from the warm-up, where they belong.
        \begin{itemize}
            \item Second click: ask what went wrong.
                \texttt{\^{}} only takes the next character, so only the 2 is raised.
            \item Braces group, exactly like the argument braces from block 1.
                \texttt{T\_\{\textbackslash mathrm\{env\}\}} on the next slide is the same rule.
            \item Order does not matter: \texttt{T\_0\^{}2} and \texttt{T\^{}2\_0} give the same output.
            \item 250\,ml of water is about $8 \cdot 10^{24}$ molecules, so the sentence is even true.
        \end{itemize}

            \outputrender{renders/03-labels-right.pdf}
    \end{overprint}
    \vfill
    \begin{exampleblock}<2-| handout:2>{Variables and Labels}
        \Large
        Variables italic, labels upright: \code|\mathrm{...}|
    \end{exampleblock}
    \begin{uncoverenv}<3-| handout:2>
        \begin{center}
            \color{black!60}
            Every label: \code|V_{\mathrm{out}}|, \code|t_{\mathrm{max}}|, \code|m_{\mathrm{water}}|
        \end{center}
    \end{uncoverenv}
    \vfill
    \note{%
        \textbf{Say it explicitly: if they keep one rule from this block, it is this one.}
        Slow down here.
        \begin{itemize}
            \item First click: ask what is wrong.
                In math mode every letter is a variable, so ``env'' is set as $e \cdot n \cdot v$:
                italic, with the spacing of three separate symbols.
            \item \texttt{T\_\{env\}} is the most common thing beginners get wrong in a report,
                and it is visible from the back of the room.
                Supervisors and reviewers notice it immediately; it marks a document as careless.
            \item Third click: it is not about this one symbol.
                Every subscript that is a word or an abbreviation is a label.
            \item The same rule makes the constant e and the d in $\mathrm{d}x$ upright (ISO 80000-2):
                \texttt{\textbackslash mathrm\{e\}}, \texttt{\textbackslash mathrm\{d\}}.
                The next slides use that.
                Many textbooks set them italic; what matters is being consistent.
        \end{itemize}
    }
\end{frame}

\begin{frame}[fragile]
    \frametitle{Common Commands}
    \begin{center}
        \large
        \renewcommand{\arraystretch}{1.5}
        \begin{tabular}{@{}l@{\hspace{3em}}l@{\hspace{3em}}l@{}}
            \toprule
            \textbf{Command} & \textbf{Output} & \\
            \midrule
            \onslide<1->\code|\frac{a}{b}|               & \onslide<1->$\displaystyle\frac{a}{b}$                  & \onslide<1->{\color{black!60}fraction} \\
            \onslide<2->\code|\sqrt{x}|                  & \onslide<2->$\sqrt{x}$                                  & \onslide<2->{\color{black!60}root} \\
            \onslide<3->\code|\alpha \tau \Delta|        & \onslide<3->$\alpha \enspace \tau \enspace \Delta$      & \onslide<3->{\color{black!60}Greek letters} \\
            \onslide<4->\code|a \cdot b|                 & \onslide<4->$a \cdot b$                                 & \onslide<4->{\color{black!60}product} \\
            \onslide<5->\code|\ln x \sin x|              & \onslide<5->$\ln x \enspace \sin x$                     & \onslide<5->{\color{black!60}function names} \\
            \onslide<6->\code|\left( \frac{a}{b} \right)| & \onslide<6->$\displaystyle\left( \frac{a}{b} \right)$  & \onslide<6->{\color{black!60}brackets that grow} \\
            \bottomrule
        \end{tabular}
    \end{center}
    \vfill
    \begin{uncoverenv}<7->
        \begin{center}
            \color{black!60}
            Other function names: \code|\operatorname{sgn}|, see the handout
        \end{center}
    \end{uncoverenv}
    \vfill
    \note{%
        Click through quickly; the handout has more of each.
        \begin{itemize}
            \item \texttt{\textbackslash frac} is the first command with two mandatory arguments: numerator, then denominator.
            \item Greek: the capital letter of the name gives the capital letter, \texttt{\textbackslash Delta} vs.\ \texttt{\textbackslash delta}.
                There is no \texttt{\textbackslash Alpha}: it would look like A.
            \item Function names (\texttt{\textbackslash ln}, \texttt{\textbackslash sin}, \texttt{\textbackslash exp}, \texttt{\textbackslash log}, \ldots)
                are upright and get the right spacing.
                Typed as plain letters, \texttt{sin} would be three italic variables.
            \item Last click: only a fixed set of names is predefined:
                the trigonometric and hyperbolic functions, \texttt{\textbackslash exp}, \texttt{\textbackslash ln}, \texttt{\textbackslash log},
                \texttt{\textbackslash lim}, \texttt{\textbackslash max}, \texttt{\textbackslash min}, \texttt{\textbackslash det} and a few more.
                No \texttt{\textbackslash sgn}, \texttt{\textbackslash erf} or \texttt{\textbackslash Tr}.
            \item For one use: \texttt{\textbackslash operatorname\{sgn\}}.
                For repeated use, once in the preamble: \texttt{\textbackslash DeclareMathOperator\{\textbackslash sgn\}\{sgn\}}, then \texttt{\textbackslash sgn} works like \texttt{\textbackslash sin}.
                Both come from \texttt{amsmath}; the handout has the details.
            \item \texttt{\textbackslash left} and \texttt{\textbackslash right} always come in pairs and size the brackets to the content.
                Plain \texttt{(} stays small, even around a tall fraction.
            \item Fractions, \texttt{\textbackslash ln} and the growing brackets are what the derivation in the exercise needs.
        \end{itemize}
    }
\end{frame}

    \uncover<2->{\outputrender{renders/03-align.pdf}}
    \vfill
    \begin{itemize}
        \item<3-> \code|&|: align here, in every line
        \item<4-> \code|\\|: new line
        \item<5-> \code|\notag|: no number for this line
    \end{itemize}
    \vfill
    \note{%
        \textbf{Predict first:} where will the lines line up?
        \begin{itemize}
            \item The \texttt{\&} goes in the same place in every line, usually right before the \texttt{=}.
            \item \texttt{align} numbers every line; \texttt{\textbackslash notag} is for a line nobody will cite.
            \item Callback: in block 2 you said \texttt{\textbackslash\textbackslash} belongs in two places,
                tables and math.
                This is one of them.
                Say ``told you''; the payoff for the earlier promise makes the next one credible.
            \item Do not teach \texttt{align*}, \texttt{gather}, \texttt{split}, \texttt{multline} or \texttt{aligned}.
                \texttt{align} plus \texttt{\textbackslash notag} covers everything they need this year;
                the handout lists the rest.
        \end{itemize}
    }
\end{frame}

    \vfill

        \begin{center}
            \Large
            Words in a formula: \code|\text{...}|
        \end{center}
    \end{uncoverenv}
    \vfill
    \note{%
        \begin{itemize}
            \item A quantity goes in math mode, a word goes in \texttt{\textbackslash text}, even mid-formula.
                Without it, ``otherwise'' would be nine italic variables without spaces, like the warm-up.
            \item \texttt{cases} only works in math mode, hence the \texttt{\textbackslash[ \textbackslash]} around it.
            \item \texttt{cases} shows \texttt{\&} and \texttt{\textbackslash\textbackslash} doing the same job as in \texttt{align}.
                Point that out.
            \item The template's version has a third branch: same shape, one more line.
            \item They will find \texttt{\textbackslash,\^{}\{\textbackslash circ\}\textbackslash mathrm\{C\}} ugly.
                Good: tell them a better way is coming at the end.
                That is the hook for \texttt{siunitx}.
        \end{itemize}
    }

\begin{frame}[fragile]

    \begin{center}
        \renewcommand{\arraystretch}{1.25}
        \begin{tabular}{@{}lll@{}}
            \toprule
            \textbf{Don't}     & \textbf{Do}                                  & \textbf{Why} \\
            \midrule
            \code|a * b|       & \code|a \cdot b|                             & \texttt{*} is convolution \\
            \code|x^10|        & \code|x^{10}|                                & more than one character \\
            \code|V_{out}|     & \code|V_{\mathrm{out}}|                      & a label, not variables \\
            \code|sin(x)|      & \code|\sin(x)|                               & upright, spaced \\
            \code|d/dx|        & \code|\frac{\mathrm{d}f}{\mathrm{d}x}|       & operator, upright \\
            \bottomrule
        \end{tabular}
    \end{center}
    \vfill
    \begin{uncoverenv}<2->
        \begin{center}
            \color{black!60}
            \code|\cdot|: product
            \qquad
            \code|\times|: cross product, dimensions
        \end{center}
    \end{uncoverenv}
    \vfill
    \note{%
        \begin{itemize}
            \item Two minutes.
                It is a reference table and it is in the handout;

            \item Pick the two that matter for \emph{this} group.
                For a mixed audience: \texttt{\textbackslash mathrm} for labels and \texttt{\textbackslash sin} for functions.
            \item The 2025 deck said ``never use \texttt{\textbackslash times}''.

        \begin{itemize}
            \item The fear of memorising hundreds of commands is the main reason people bounce off \LaTeX{}.
                This kills it.
            \item Second click, the lists: Overleaf's is short and a good start,
                the OEIS one has long tables,
                and the Comprehensive Symbol List has every symbol of every package.
                Search it, do not read it.
                \texttt{texdoc comprehensive} opens it on a local installation.
            \item All of them are in the handout and in a comment in the template.
            \item The same is true of everything today: you look it up until you stop needing to.
        \end{itemize}
    }
